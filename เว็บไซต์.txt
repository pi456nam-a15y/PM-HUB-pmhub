        function formatBytes(bytes) {
            if (bytes === 0) return '0 B';
            const k = 1024;
            const sizes = ['B', 'KB', 'MB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
        }

        function makeVarName(style = 'english', index = 1) {
            if (style === 'barcode') {
                const parts = ['I', 'l', '1'];
                let res = 'I';
                for(let i = 0; i < 8; i++) res += parts[Math.floor(Math.random()*parts.length)];
                return res + '_' + index;
            } else if (style === 'short') {
                return `_${String.fromCharCode(97 + (index % 26))}${index}`;
            } else {
                const words = ['DataStr', 'MapTable', 'ByteBuf', 'SubFunc', 'CharFunc', 'ExecScript', 'LoadFunc', 'Runner'];
                return (words[index % words.length] || 'v') + Math.floor(Math.random() * 899 + 100);
            }
        }

        // Lightweight Obfuscation Core Engine
        function generateLightweightLua(sourceText, preset, alphaStyle) {
            const alpha = ALPHABETS[alphaStyle] || ALPHABETS.uppercase;
            lastAlphabetUsed = alpha;

            const bytes = new TextEncoder().encode(sourceText);
            
            let letterPairs = [];
            for (let i = 0; i < bytes.length; i++) {
                const b = bytes[i];
                const hi = (b >> 4) & 0x0F;
                const lo = b & 0x0F;
                letterPairs.push(alpha[hi] + alpha[lo]);
            }
            const encodedString = letterPairs.join('');
            lastEncodedLetters = encodedString;

            let luaMapEntries = [];
            for (let i = 0; i < 16; i++) {
                luaMapEntries.push(`["${alpha[i]}"]=${i}`);
            }
            const mapTableStr = `{${luaMapEntries.join(',')}}`;

            let varStyle = 'short';
            if (preset === 'english_vars') varStyle = 'english';
            if (preset === 'barcode') varStyle = 'barcode';

            const vData = makeVarName(varStyle, 1);
            const vMap = makeVarName(varStyle, 2);
            const vOut = makeVarName(varStyle, 3);
            const vSub = makeVarName(varStyle, 4);
            const vChar = makeVarName(varStyle, 5);
            const vI = makeVarName(varStyle, 6);
            const vH = makeVarName(varStyle, 7);
            const vL = makeVarName(varStyle, 8);
            const vScript = makeVarName(varStyle, 9);
            const vExec = makeVarName(varStyle, 10);

            let lua = `-- [[ PMHUB GITHUB OBFUSCATED ]]\n`;
            lua += `local ${vData} = "${encodedString}"\n`;
            lua += `local ${vMap} = ${mapTableStr}\n`;
            lua += `local ${vOut} = {}\n`;
            lua += `local ${vSub} = string.sub\n`;
            lua += `local ${vChar} = string.char\n\n`;
            lua += `for ${vI} = 1, #${vData}, 2 do\n`;
            lua += `    local ${vH} = ${vMap}[${vSub}(${vData}, ${vI}, ${vI})]\n`;
            lua += `    local ${vL} = ${vMap}[${vSub}(${vData}, ${vI} + 1, ${vI} + 1)]\n`;
            lua += `    if ${vH} and ${vL} then\n`;
            lua += `        ${vOut}[#${vOut} + 1] = ${vChar}(${vH} * 16 + ${vL})\n`;
            lua += `    end\n`;
            lua += `end\n\n`;
            lua += `local ${vScript} = table.concat(${vOut})\n`;
            lua += `local ${vExec} = assert((loadstring or load)(${vScript}), "Error loading script")\n`;
            lua += `return ${vExec}()`;

            return lua;
        }

        function processObfuscation() {
            const inputCode = luaInput.value;
            if (!inputCode.trim()) {
                showToast("กรุณากรอกโค้ด Lua ก่อนทำการปกปิด", false);
                return;
            }

            const preset = document.getElementById('obfPreset').value;
            const alphaStyle = document.getElementById('alphabetStyle').value;

            const btn = document.getElementById('btnObfuscate');
            btn.disabled = true;
            btn.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i> <span>กำลังประมวลผล...</span>`;

            setTimeout(() => {
                try {
                    const obfuscatedLua = generateLightweightLua(inputCode, preset, alphaStyle);

                    luaOutput.textContent = obfuscatedLua;
                    luaOutput.classList.remove('hidden');
                    outputPlaceholder.classList.add('hidden');
                    statusBadge.classList.remove('hidden');
                    outputStats.classList.remove('hidden');

                    const origSize = new Blob([inputCode]).size;
                    const newSize = new Blob([obfuscatedLua]).size;
                    const ratio = (newSize / Math.max(origSize, 1)).toFixed(1);

                    outputSize.textContent = formatBytes(newSize);
                    ratioStat.textContent = `${ratio}x`;

                    // Generate GitHub Loadstring if credentials provided
                    const u = ghUser.value.trim();
                    const r = ghRepo.value.trim();
                    const f = ghFile.value.trim() || 'script.lua';

                    if (u && r) {
                        const rawUrl = `https://raw.githubusercontent.com/${u}/${r}/main/${f}`;
                        const loadstrCmd = `loadstring(game:HttpGet("${rawUrl}"))()`;
                        loadstringContent.textContent = loadstrCmd;

                    } else {
                        loadstringBox.classList.add('hidden');
                    }

                    showToast("ปกปิดสคริปต์และสร้างคำสั่ง GitHub สำเร็จ!");

                } catch (err) {
                    showToast(err.message || "เกิดข้อผิดพลาดในการปกปิดโค้ด", false);
                } finally {
                    btn.disabled = false;
                    btn.innerHTML = `<i class="fa-solid fa-lock text-base"></i> <span>🔒 ล็อคและปกปิดสคริปต์ (พร้อมสร้าง GitHub Loadstring)</span>`;
                }
            }, 40);

        function testDecodeInBrowser() {
            if (!lastEncodedLetters || !lastAlphabetUsed) {
                showToast("ยังไม่มีโค้ดที่ปกปิดให้ทดสอบ", false);
                return;
            }

            try {
                let map = {};
                for (let i = 0; i < lastAlphabetUsed.length; i++) {
                    map[lastAlphabetUsed[i]] = i;
                }

                let bytes = [];
                for (let i = 0; i < lastEncodedLetters.length; i += 2) {
                    let hChar = lastEncodedLetters[i];
                    let lChar = lastEncodedLetters[i + 1];
                    let h = map[hChar];
                    let l = map[lChar];
                    if (h !== undefined && l !== undefined) {
                        bytes.push((h << 4) | l);
                    }
                }

                const decodedText = new TextDecoder().decode(new Uint8Array(bytes));
                testDecodeContent.textContent = decodedText;
                testDecodeBox.classList.remove('hidden');
                testDecodeBox.scrollIntoView({ behavior: 'smooth' });
                showToast("ทดสอบถอดรหัสสำเร็จ! โค้ดถูกต้อง 100%");
            } catch (e) {
                showToast("การถอดรหัสล้มเหลว: " + e.message, false);
            }
        }

        function downloadLuaFile() {

            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = filename;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);

            showToast(`📥 ดาวน์โหลดไฟล์ ${filename} สำเร็จ!`);
        }
